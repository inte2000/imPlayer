cmake_minimum_required(VERSION 3.21)
project(imPlayer VERSION 0.1.2 LANGUAGES C CXX)

if(NOT MSVC)
    message(FATAL_ERROR "This project is intended to be built with MSVC")
endif()

if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(IPLAYER_ARCH x64)
else()
    set(IPLAYER_ARCH x86)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
include(msvc_options)
include(CTest)

set(THIRDLIB_ROOT "${PROJECT_SOURCE_DIR}/thirdparty")
set(IPLAYER_DEBUG_OUTPUT_DIR "${PROJECT_SOURCE_DIR}/build/debug/bin")
set(IPLAYER_RELEASE_OUTPUT_DIR "${PROJECT_SOURCE_DIR}/build/Release/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(kind RUNTIME LIBRARY ARCHIVE)
    set(CMAKE_${kind}_OUTPUT_DIRECTORY_DEBUG "${IPLAYER_DEBUG_OUTPUT_DIR}")
    set(CMAKE_${kind}_OUTPUT_DIRECTORY_RELEASE "${IPLAYER_RELEASE_OUTPUT_DIR}")
endforeach()

add_subdirectory(src/core)
add_subdirectory(src/decoder)
add_subdirectory(src/player)
if(BUILD_TESTING)
    add_subdirectory(src/test)
endif()
